decl
    int x, y;
    int swap(int *p, int *q);
enddecl

int swap(int *p, int *q)
{
    decl
        int temp;
    enddecl

    begin
        temp = *p;
        *p = *q;
        *q = temp;
        return 0;
    end
}

int main()
{
    decl
        int status;
    enddecl

    begin
        x = 10;
        y = 20;
        status = swap(&x, &y);
        write(x);
        write(y);
        return 0;
    end
}
