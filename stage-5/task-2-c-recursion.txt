decl
    int fact(int n);
enddecl

int fact(int n)
{
    decl
        int answer;
    enddecl

    begin
        if (n == 0) then
            answer = 1;
        else
            answer = n * fact(n - 1);
        endif;
        return answer;
    end
}

int main()
{
    decl
        int result;
    enddecl

    begin
        result = fact(5);
        write(result);
        return 0;
    end
}
