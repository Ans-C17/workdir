decl
    int sumValues();
enddecl

int sumValues() {
    decl
        int values[4], total;
    enddecl

    begin
        values[0] = 10;
        values[1] = 20;
        values[2] = 30;
        values[3] = 40;
        total = values[0] + values[1] + values[2] + values[3];
        return total;
    end
}

int main() {
    decl
        int result;
    enddecl

    begin
        result = sumValues();
        write(result);
        return 0;
    end
}
